# GDB Breakpoint Configuration for torch-spyre Educational Walkthrough
#
# Usage:
#   1. Compile torch-spyre with debug symbols:
#      CFLAGS="-g -O0" CXXFLAGS="-g -O0" pip install -e .
#
#   2. Start GDB:
#      gdb --args python examples/mul_debug.py
#
#   3. Load this file:
#      (gdb) source examples/gdb_breakpoints.txt
#
#   4. Run:
#      (gdb) run

# ============================================================================
# MEMORY ALLOCATION BREAKPOINTS
# ============================================================================

# Break when allocating device memory
break spyre::SpyreAllocator::allocate
commands
  silent
  printf "\n========================================\n"
  printf "BREAKPOINT: SpyreAllocator::allocate\n"
  printf "========================================\n"
  printf "Allocating %zu bytes on Spyre device\n", nbytes
  printf "Current device: %d\n", curr_device.index()
  printf "\nPress 'c' to continue...\n"
  printf "========================================\n\n"
end

# Break when computing tensor layout

# ============================================================================
# DATA TRANSFER BREAKPOINTS
# ============================================================================

# Break when copying host to device
break spyre::copy_host_to_device
commands
  silent
  printf "\n========================================\n"
  printf "BREAKPOINT: copy_host_to_device\n"
  printf "========================================\n"
  printf "Transferring tensor from CPU to Spyre\n"
  printf "Source tensor shape: "
  call self.sizes()
  printf "\nDestination tensor shape: "
  call dst.sizes()
  printf "\n\nThis will:\n"
  printf "  1. Create DMA graph\n"
  printf "  2. Apply layout transformation\n"
  printf "  3. Execute DMA transfer\n"
  printf "\nPress 'c' to continue...\n"
  printf "========================================\n\n"
end
